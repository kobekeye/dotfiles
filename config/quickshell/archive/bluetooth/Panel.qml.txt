import Quickshell
import QtQuick

import Quickshell.Wayland

PanelWindow {
    id: root

    property var bluetooth

    WlrLayershell.namespace: "bluetooth-panel"
    implicitWidth: 340
    implicitHeight: content.implicitHeight + 32
    property color panelColor: "#e6262d2c"
    property color textPrimary: "#eeeae1"
    property color textSecondary: "#aaa9a4"
    property color accentColor: "#a7835c"
    property color inactiveColor: "#30ffffff"
    property color hoverColor: "#20ffffff"
    color: "transparent"
    property string uiFont: "Geist"
    // press Esc can hide
    focusable: true
    Shortcut {
        sequence: "Escape"

        onActivated: {
            root.visible = false
        }
    }

    anchors {
        top: true
        right: true
    }
    margins {
        top: 0
        right: 210
    }
    Rectangle{
        anchors.fill: parent
        color: root.panelColor
        radius: 16
        border.width: 1
        border.color: "#18ffffff"

        Column{
            id: content
            anchors{
                top: parent.top
                left: parent.left
                right: parent.right

                topMargin: 16
                leftMargin: 16
                rightMargin: 16
            }
            spacing: 15

            // shows whether bluetooth on or off
            Item {
                width: parent.width
                height: 28

                // the bluetooth title
                Text {
                    anchors {
                        left: parent.left
                        verticalCenter: parent.verticalCenter
                    }

                    text: "Bluetooth"
                    color: root.textPrimary

                    font {
                        family: root.uiFont
                        pixelSize: 17
                        weight: Font.Medium
                    }
                }

                // on/off switch
                Rectangle {
                    id: powerSwitch

                    property bool checked:
                        bluetooth.adapter
                        && bluetooth.adapter.enabled

                    anchors {
                        right: parent.right
                        verticalCenter: parent.verticalCenter
                    }

                    width: 42
                    height: 24
                    radius: 12

                    color: checked
                        ? root.accentColor
                        : root.inactiveColor

                    // the switch ball
                    Rectangle {
                        width: 18
                        height: 18
                        radius: 9

                        y: 3

                        x: powerSwitch.checked
                            ? parent.width - width - 3
                            : 3

                        color: root.textPrimary

                        // the switch ball animation
                        Behavior on x {
                            NumberAnimation {
                                duration: 120
                            }
                        }
                    }


                    MouseArea {
                        anchors.fill: parent

                        cursorShape: Qt.PointingHandCursor
                        onClicked: bluetooth.togglePower()
                    }
                }
            }

            // bluetooth list devices

            Text {
                text: "My Devices"
                color: root.textSecondary
                font {
                    family: root.uiFont
                    pixelSize: 12
                    weight: Font.Medium
                }
            }

                width: parent.width
                height: Math.min(contentHeight, 140)
                clip: true
                model: bluetooth.adapter
                    ? bluetooth.adapter.devices
                    : null

                delegate: Rectangle {
                    required property var modelData
                    property var device: modelData
                    property bool shouldShow:
                        device.deviceName
                        && device.deviceName.length > 0
                        && device.paired

                    width: ListView.view.width
                    height: shouldShow ? 45 : 0
                    visible: shouldShow
                    radius: 10

                    color: mouse.containsMouse
                        ? root.hoverColor
                        : "transparent"

                    Image {
                        id: deviceIcon
                        width: 20
                        height: 20

                        source: Quickshell.iconPath(device.icon, "network-bluetooth")

                        anchors {
                            left: parent.left
                            leftMargin: 10
                            verticalCenter: parent.verticalCenter
                        }
                    }
                    Column {
                        anchors {
                            left: deviceIcon.right
                            leftMargin: 10
                            verticalCenter: parent.verticalCenter
                        }

                        spacing: 2

                        Text {
                            text: device.name
                            color: root.textPrimary
                            font {
                                family: root.uiFont
                                pixelSize: 12
                            }
                        }

                        Text {
                            text: {
                                if (device.state === BluetoothDeviceState.Connected)
                                    return "Connected"

                                if (device.state === BluetoothDeviceState.Connecting)
                                    return "Connecting..."

                                if (device.state === BluetoothDeviceState.Disconnecting)
                                    return "Disconnecting..."

                                return "Paired"
                            }

                            color: root.textSecondary
                            font {
                                family: root.uiFont
                                pixelSize: 10
                            }
                        }
                    }

                    Text {
                        anchors {
                            right: parent.right
                            rightMargin: 10
                            verticalCenter: parent.verticalCenter
                        }

                        visible: device.batteryAvailable
                        text: Math.round(device.battery * 100) + "%"

                        color: root.textSecondary
                        font {
                            family: root.uiFont
                            pixelSize: 11
                        }
                    }

                    MouseArea {
                        id: mouse
                        anchors.fill: parent

                        hoverEnabled: true
                        cursorShape: Qt.PointingHandCursor
                        enabled:
                            device.state !== BluetoothDeviceState.Connecting
                            && device.state !== BluetoothDeviceState.Disconnecting

                        onClicked: {
                            if (device.connected)
                                device.disconnect()
                            else
                                device.connect()
                        }
                    }
                }

            }

            // available devices & scan
            Item {
                width: parent.width
                height: 24

                // available devices title
                Text {
                    anchors {
                        left: parent.left
                        verticalCenter: parent.verticalCenter

                    text: "Available Devices"

                    color: root.textSecondary

                    font {
                        family: root.uiFont
                        pixelSize: 12
                        weight: Font.Medium

                        right: parent.right
                        verticalCenter: parent.verticalCenter

                    width: 58
                    height: 24
                    radius: 12

                        && bluetooth.adapter.discovering
                            ? root.accentColor
                            : root.inactiveColor


                    Text {
                        anchors.centerIn: parent

                        text:
                            bluetooth.adapter
                            && bluetooth.adapter.discovering
                                ? "Stop"
                                : "Scan"


                        color: root.textPrimary

                        font {
                            family: root.uiFont
                            pixelSize: 11
                            weight: Font.Medium
                        }
                    }


                    MouseArea {
                        anchors.fill: parent
                        cursorShape: Qt.PointingHandCursor

                        onClicked: bluetooth.scan()
                    }
                }
            }

                        && !device.paired

                    width: ListView.view.width
                    height: shouldShow ? 45 : 0
                    radius: 10

                    color: mouse.containsMouse

                    Image {
                        id: deviceIcon
                        width: 20
                        height: 20

                        source: Quickshell.iconPath(device.icon, "network-bluetooth")

                        anchors {
                            left: parent.left
                            leftMargin: 10
                            verticalCenter: parent.verticalCenter
                        }
                    }
                    Column{
                        anchors {
                            left: deviceIcon.right
                            leftMargin: 10
                            verticalCenter: parent.verticalCenter
                        }
                        spacing: 2
                        Text {
                            text: device.name
                            color: root.textPrimary
                            font {
                                family: root.uiFont
                                pixelSize: 12
                            }
                        }

                        Text {
                            text: device.pairing
                                ? "Pairing..."
                                : "Not paired"

                            color: root.textSecondary
                            font {
                                family: root.uiFont
                                pixelSize: 10
                            }
                        }
                    }

                    MouseArea {
                        id: mouse
                        hoverEnabled: true
                        cursorShape: Qt.PointingHandCursor
                        anchors.fill: parent
                        enabled: !device.pairing
                        onClicked: {
                            device.pair()
                        }
                    }
                }
            }
